-- ============================
-- Companion Cavalry (Knight)
-- upped to knight
-- ==============================

INSERT INTO Civilization_UnitClassOverrides
	(CivilizationType, UnitClassType, UnitType)
SELECT
	CivilizationType, 'UNITCLASS_KNIGHT', 'UNIT_GREEK_COMPANIONCAVALRY'
FROM Civilization_JFD_CultureTypes WHERE CultureType = 'CULTURE_JFD_CLASSICAL';

CREATE TRIGGER ClassicalEquestrian
AFTER INSERT ON Civilization_JFD_CultureTypes WHEN NEW.CultureType = 'CULTURE_JFD_CLASSICAL'
BEGIN
	INSERT INTO Civilization_UnitClassOverrides
		(CivilizationType, UnitClassType, UnitType)
	SELECT
		NEW.CivilizationType, 'UNITCLASS_KNIGHT', 'UNIT_GREEK_COMPANIONCAVALRY';
END;

UPDATE Units SET 
Combat = 22,
Cost = 150,
FaithCost = 275,
PrereqTech = 'TECH_CURRENCY',
ObsoleteTech = 'TECH_COMBUSTION',
MinorCivGift = 0,
GoodyHutUpgradeUnitClass = 'UNITCLASS_LANCER',
Class = 'UNITCLASS_KNIGHT'
WHERE Type = 'UNIT_GREEK_COMPANIONCAVALRY';

UPDATE Unit_ClassUpgrades SET UnitClassType = 'UNITCLASS_LANCER' WHERE UnitType = 'UNIT_GREEK_COMPANIONCAVALRY';
